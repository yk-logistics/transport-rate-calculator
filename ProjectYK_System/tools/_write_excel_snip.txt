1169|def interleaved_matched_unmatched_rows_html(
1170|    trips: list[Trip],
1171|    unmatched: list[tuple[str, Leg, str]],
1172|    trip_row_cb: Callable[[Trip], str],
1173|    *,
1174|    plate: str | None = None,
1175|    include_plate_link: bool = True,
1176|    include_plate_column: bool = True,
1177|) -> str:
1178|    """Sort matched (by Dest_In time) and unmatched (by leg time) into one timeline."""
1179|    rows: list[tuple[datetime, tuple[Any, ...], str]] = []
1180|    for t in trips:
1181|        if plate is not None and t.plate != plate:
1182|            continue
1183|        rows.append((t.d_in, (0, t.plate, t.d_row or "", t.o_row or ""), trip_row_cb(t)))
1184|    for src, leg, _mp in unmatched:
1185|        if plate is not None and leg.plate != plate:
1186|            continue
1187|        um_html = unmatched_merged_trip_one_row_html(
1188|            src,
1189|            leg,
1190|            include_plate_link=include_plate_link,
1191|            include_plate_column=include_plate_column,
1192|        )
1193|        kind = 1 if src == "Origin" else 2
1194|        rows.append((leg.t_in, (kind, leg.plate, src, leg.row_no), um_html))
1195|    rows.sort(key=lambda x: (x[0], x[1]))
1196|    return "".join(r[2] for r in rows)
